#pragma once
// Standalone assertion adapter for the planner's ordinary GoogleTest test source.
// The repository C++ runner uses real GoogleTest, not this header.
#include <functional>
#include <iostream>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>
namespace standalone_test {
using Case = std::pair<std::string, std::function<void()>>;
inline std::vector<Case> &cases() {
  static std::vector<Case> value;
  return value;
}
inline bool add(const char *name, std::function<void()> function) {
  cases().emplace_back(name, std::move(function));
  return true;
}
inline void check(bool passed, const char *expression, const char *file, int line) {
  if (!passed)
    throw std::runtime_error(std::string(file) + ":" + std::to_string(line) + ": " + expression);
}
inline int run() {
  size_t failed = 0;
  for (const auto &[name, function] : cases()) {
    try {
      function();
      std::cout << "PASS " << name << '\n';
    } catch (const std::exception &error) {
      ++failed;
      std::cerr << "FAIL " << name << ": " << error.what() << '\n';
    }
  }
  std::cout << "Planner cases: " << cases().size() - failed << " passed, " << failed << " failed\n";
  return failed == 0 ? 0 : 1;
}
}  // namespace standalone_test
#define TEST(suite, name) \
  static void suite##_##name(); \
  static const bool suite##_##name##_registered = standalone_test::add(#suite "." #name, suite##_##name); \
  static void suite##_##name()
#define ASSERT_TRUE(x) standalone_test::check(static_cast<bool>(x), #x, __FILE__, __LINE__)
#define ASSERT_FALSE(x) ASSERT_TRUE(!(x))
#define EXPECT_TRUE(x) ASSERT_TRUE(x)
#define EXPECT_FALSE(x) ASSERT_FALSE(x)
#define EXPECT_EQ(a, b) ASSERT_TRUE((a) == (b))
#define ASSERT_EQ(a, b) EXPECT_EQ(a, b)
#define EXPECT_LE(a, b) ASSERT_TRUE((a) <= (b))
#define ASSERT_LE(a, b) EXPECT_LE(a, b)
#define EXPECT_LT(a, b) ASSERT_TRUE((a) < (b))
#define EXPECT_GE(a, b) ASSERT_TRUE((a) >= (b))

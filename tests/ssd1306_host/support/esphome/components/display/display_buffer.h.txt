#pragma once
#include <functional>
#include "esphome/core/component.h"
#include "esphome/core/hal.h"
#include "esphome/core/helpers.h"
namespace esphome {
struct Color {
  bool on;
  bool is_on() const { return this->on; }
  static const Color BLACK;
  static const Color WHITE;
};
inline const Color Color::BLACK{false};
inline const Color Color::WHITE{true};
namespace display {
enum class DisplayType { DISPLAY_TYPE_BINARY };
class Display : public Component {
 public:
  virtual void fill(Color) {}
  struct Clipping {
    bool is_set() const { return false; }
  };
  Clipping get_clipping() const { return {}; }
};
class DisplayBuffer : public Display {
 public:
  ~DisplayBuffer() override { RAMAllocator<uint8_t>().deallocate(this->buffer_, 0); }
  virtual DisplayType get_display_type() { return DisplayType::DISPLAY_TYPE_BINARY; }
  void clear() { this->fill(Color::BLACK); }
  void draw_pixel_at(int x, int y, Color c = Color::WHITE) { this->draw_absolute_pixel_internal(x, y, c); }
  void set_writer(std::function<void(Display &)> fn) { this->writer_ = std::move(fn); }
  uint8_t *test_buffer() { return this->buffer_; }
  size_t test_render_calls() const { return this->render_calls_; }

 protected:
  virtual void draw_absolute_pixel_internal(int, int, Color) {}
  virtual int get_width_internal() { return 0; }
  virtual int get_height_internal() { return 0; }
  void init_internal_(uint32_t n) {
    this->buffer_ = RAMAllocator<uint8_t>().allocate(n);
    if (this->buffer_ == nullptr)
      return;
    this->clear();
  }
  void do_update_() {
    ++this->render_calls_;
    if (this->writer_)
      this->writer_(*this);
  }
  uint8_t *buffer_{nullptr};
  std::function<void(Display &)> writer_;
  size_t render_calls_{0};
};
}  // namespace display
}  // namespace esphome

#pragma once
#include <array>
#include <cstring>
#include "esphome/core/helpers.h"
#include "esphome/core/log.h"
#define LOG_I2C_DEVICE(obj) (void) (obj)
namespace esphome::i2c {
enum ErrorCode { ERROR_OK = 0, ERROR_NOT_ACKNOWLEDGED = 2 };
class I2CBus {
 public:
  virtual ~I2CBus() = default;
  virtual ErrorCode write_readv(uint8_t, const uint8_t *, size_t, uint8_t *, size_t) = 0;
};
class I2CDevice {
 public:
  void set_i2c_address(uint8_t address) { this->address_ = address; }
  void set_i2c_bus(I2CBus *bus) { this->bus_ = bus; }
  ErrorCode write(const uint8_t *data, size_t n) const {
    return this->bus_->write_readv(this->address_, data, n, nullptr, 0);
  }
  ErrorCode write_register(uint8_t reg, const uint8_t *data, size_t n) const {
    uint8_t local[17];
    RAMUniquePtr<uint8_t[]> dynamic;
    uint8_t *buffer = local;
    if (n > 16) {
      dynamic = RAMAllocator<uint8_t>().make_unique_array_for_overwrite(n + 1);
      if (!dynamic)
        return ERROR_NOT_ACKNOWLEDGED;
      buffer = dynamic.get();
    }
    buffer[0] = reg;
    if (n != 0)
      std::memcpy(buffer + 1, data, n);
    return this->write(buffer, n + 1);
  }
  bool write_bytes(uint8_t reg, const uint8_t *data, uint8_t n) const {
    return this->write_register(reg, data, n) == ERROR_OK;
  }
  bool write_byte(uint8_t reg, uint8_t value) const { return this->write_bytes(reg, &value, 1); }

 protected:
  I2CBus *bus_{nullptr};
  uint8_t address_{0x3C};
};
}  // namespace esphome::i2c

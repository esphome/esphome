#pragma once
#include <cstdint>
namespace esphome {
using LogString = char;
namespace setup_priority {
inline constexpr float PROCESSOR = 400.0f;
}
class Component {
 public:
  virtual ~Component() = default;
  virtual void setup() {}
  virtual void dump_config() {}
  virtual void update() {}
  virtual void loop() {}
  virtual float get_setup_priority() const { return 0; }
  void mark_failed(const LogString * = nullptr) {
    this->failed_ = true;
    this->loop_enabled_ = false;
  }
  bool is_failed() const { return this->failed_; }
  void status_set_warning(const LogString * = nullptr) { this->warning_ = true; }
  void status_clear_warning() { this->warning_ = false; }
  bool status_has_warning() const { return this->warning_; }
  void enable_loop() { this->loop_enabled_ = true; }
  void disable_loop() { this->loop_enabled_ = false; }
  bool test_loop_enabled() const { return this->loop_enabled_; }

 protected:
  bool failed_{false};
  bool warning_{false};
  bool loop_enabled_{true};
};
}  // namespace esphome
